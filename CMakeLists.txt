cmake_minimum_required(VERSION 3.20)
project(MyTorrent VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "MyTorrent 0.1 supports Windows.")
endif()
add_library(torrent_core
    src/bencode.cpp src/metainfo.cpp src/storage.cpp src/net.cpp src/tracker.cpp
    src/peer.cpp src/scheduler.cpp src/transfer.cpp src/discovery.cpp src/console_interrupt.cpp)
target_include_directories(torrent_core PUBLIC src)
target_compile_features(torrent_core PUBLIC cxx_std_20)
set_target_properties(torrent_core PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(torrent_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
target_link_libraries(torrent_core PRIVATE bcrypt ws2_32 winhttp)

add_executable(mytorrent src/main.cpp)
target_link_libraries(mytorrent PRIVATE torrent_core)
if(MINGW)
    target_link_options(mytorrent PRIVATE -municode)
endif()

function(enable_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
enable_warnings(torrent_core)
enable_warnings(mytorrent)

include(CTest)
if(BUILD_TESTING)
    add_executable(torrent_tests tests/unit.cpp)
    target_link_libraries(torrent_tests PRIVATE torrent_core)
    enable_warnings(torrent_tests)
    add_test(NAME unit COMMAND torrent_tests)
    find_program(POWERSHELL_EXE NAMES pwsh powershell REQUIRED)
    add_test(NAME loopback COMMAND "${POWERSHELL_EXE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/loopback.ps1" -Exe "$<TARGET_FILE:mytorrent>")
    add_test(NAME parallel COMMAND "${POWERSHELL_EXE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/parallel.ps1" -Exe "$<TARGET_FILE:mytorrent>")
    add_test(NAME discovery COMMAND "${POWERSHELL_EXE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/discovery.ps1" -Exe "$<TARGET_FILE:mytorrent>")
    set_tests_properties(loopback parallel discovery PROPERTIES TIMEOUT 75)
endif()
